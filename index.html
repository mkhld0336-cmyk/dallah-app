// ==================================================================== // ⚡ GAME ENGINE & CARD ACTIVATION SYSTEM (ring-game core) // ==================================================================== // حالة اللاعب والبيانات الأساسية let player = { name: "القائد", level: 1, balance: 0, shield: false, score: 0, inventory: [] }; // تهيئة اللعبة عند التحميل window.onload = function() { console.log("⚡ تم تشغيل محرك اللعبة بنجاح..."); initGameUI(); }; // إنشاء واجهة اللعبة داخل المكون الرئيسي function initGameUI() { const gameContainer = document.getElementById('game'); if (!gameContainer) return; gameContainer.innerHTML = `

🎮 لوحة تحكم اللعبة

الرصيد: ${player.balance} ريال

المستوى: ${player.level}

الدرع: ${player.shield ? 'نشط 🛡️' : 'غير غير نشط ❌'}

النقاط: ${player.score}

🔑 أدخل كود شحن بطاقة دله


`; } // دالة معالجة الأكواد المدخلة من متجر دله function submitCardCode() { const inputField = document.getElementById('card-code-input'); const code = inputField.value.trim().toUpperCase(); if (code === "") { showMessage("الرجاء إدخال كود بطاقة الشحن!", "orange"); return; } receiveCharge(code); inputField.value = ""; } // محرك تفعيل البطاقات ومنح المكافآت الحصرية function receiveCharge(chargeKey) { // 1. البطاقة البرونزية (35 ريال) if (chargeKey === "BRONZE35" || chargeKey.startsWith("BRZ-")) { player.balance += 35; player.score += 500; player.shield = true; player.inventory.push("درع بداية اللعبة"); showMessage("✅ تم تفعيل البطاقة البرونزية! حصلت على 35 ريال رصيد + 500 نقطة + درع الحماية!", "#cd7f32"); } // 2. البطاقة الفضية (75 ريال) else if (chargeKey === "SILVER75" || chargeKey.startsWith("SLV-")) { player.balance += 75; player.score += 1500; player.level += 1; player.shield = true; player.inventory.push("سلاح المحارب المطور"); showMessage("⚡ تم تفعيل البطاقة الفضية! حصلت على 75 ريال رصيد + ترقية مستوى + سلاح فضي!", "#c0c0c0"); } // 3. البطاقة الذهبية الملكية (150 ريال) else if (chargeKey === "GOLD150" || chargeKey.startsWith("GLD-")) { player.balance += 150; player.score += 5000; player.level += 3; player.shield = true; player.inventory.push("الحزمة الأسطورية الشاملة"); showMessage("🔥 تم تفعيل البطاقة الذهبية الملكية! فتح كامل الميزات + 150 ريال + 3 مستويات!", "#ffc107"); } else { showMessage("❌ الكود غير صحيح أو مستخدم من قبل، يرجى التأكد من متجر دله.", "red"); return; } // تحديث الواجهة فوراً updateUI(); } // تحديث بيانات الصفحة function updateUI() { document.getElementById('player-balance').innerText = player.balance; document.getElementById('player-level').innerText = player.level; document.getElementById('player-shield').innerText = player.shield ? 'نشط 🛡️' : 'غير نشط ❌'; document.getElementById('player-score').innerText = player.score; } // عرض رسائل النظام function showMessage(msg, color) { const msgElement = document.getElementById('system-message'); msgElement.innerText = msg; msgElement.style.color = color; }
